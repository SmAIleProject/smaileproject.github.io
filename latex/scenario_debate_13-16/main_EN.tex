% --- Activities ---
\section{Topics for Debate}
\subsection{AI in Education}
\begin{itemize}
    \item \textbf{Surveillance:} "Should AI be used to monitor students in schools?" (Safety vs. Privacy).
    \item \textbf{Tools:} "Should schools teach students how to use AI tools like ChatGPT?".
\end{itemize}

\subsection{AI and Ethics}
\begin{itemize}
    \item \textbf{Healthcare:} "Should decisions about patient care be made by AI?".
    \item \textbf{Employment:} "Should AI be used to decide who gets a job?".
\end{itemize}

\subsection{AI in Everyday Life}
\begin{itemize}
    \item \textbf{Social Media:} "Should AI control what we see on social media?" (Algorithms vs. Free Choice).
    \item \textbf{Transport:} "Should you trust an autonomous car?".
\end{itemize}

\section{The Debate Structure}
\begin{itemize}
    \item \textbf{Constructive Speeches:} Teams present their case (2 min).
    \item \textbf{Cross-Examination:} Direct questioning of opponents (3 min).
    \item \textbf{Rebuttal:} Refuting arguments (5 min).
\end{itemize}

\section{Reflection and Evaluation}
\begin{itemize}
    \item \textbf{Judges:} Score based on Arguments (30\%), Evidence (25\%), Delivery (20\%), and Rebuttals (15\%).
    \item \textbf{Audience:} Participate via Padlet to vote for the most convincing team.
    \item \textbf{Self-Eval:} Students reflect on their own performance.
\end{itemize}
