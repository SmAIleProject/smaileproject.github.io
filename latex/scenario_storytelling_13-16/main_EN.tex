% --- Activities ---
\section{Research and Learning}
\subsection{Critical Explorer Profiles (Jigsaw)}
\textbf{Goal:} Beyond the "Hero" narrative.
\begin{itemize}
    \item \textbf{Task:} Groups research Columbus, Da Gama, Magellan, or Vespucci.
    \item \textbf{Focus:} Sponsors, motivations, and impact on indigenous populations.
    \item \textbf{Critical Reflection:} How is the explorer remembered? Who tells the story?
\end{itemize}

\subsection{Who is at the Center? (Maps)}
\textbf{Goal:} Analyzing map bias.
\begin{itemize}
    \item \textbf{Visuals:} Compare Mercator vs. Peters projections.
    \item \textbf{Discussion:} "Why is Europe in the center?" "How do maps distort reality?"
\end{itemize}

\subsection{The Columbian Exchange}
\textbf{Goal:} Understanding global trade.
\begin{itemize}
    \item \textbf{Hook:} Show items (Tomato, Chocolate). "These didn't exist in Europe before 1492!"
    \item \textbf{Task:} Trace the origin and impact of goods like potatoes or tobacco.
\end{itemize}

\section{Creative Application}
\subsection{The Diary of an Indigenous Observer}
\textbf{Task:} Shift the perspective.
\begin{itemize}
    \item Write a script for a podcast from the viewpoint of an indigenous person meeting explorers.
    \item Use AI tools to generate background imagery or sound effects.
\end{itemize}

\subsection{Interactive StoryMap}
\textbf{Task:} Visualizing globalization.
\begin{itemize}
    \item Use \textbf{StoryMapJS} to build a narrative map showing the journey of a specific commodity.
    \item Include images, text, and historical facts.
\end{itemize}
