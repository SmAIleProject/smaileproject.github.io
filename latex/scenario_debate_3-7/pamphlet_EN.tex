\pamphletheading{Introduction}
Can a robot be a teacher?

In this workshop, young children (4-7) listen to a robot story, share their own opinions, and begin to think about fairness, kindness, and what makes people different from machines.

\vfill\null
\pamphletheading{Key Goals}
\begin{tcolorbox}[colback=SmaileLightPurple, colframe=SmailePurple, boxrule=0.5pt, arc=2mm]
\small
\begin{itemize}[leftmargin=*]
    \item \textbf{Think:} About fairness and kindness.
    \item \textbf{Speak:} Share simple opinions.
    \item \textbf{Listen:} Respect different ideas.
    \item \textbf{Create:} Design a robot teacher.
\end{itemize}
\end{tcolorbox}

\columnbreak
\pamphletheading{Resources}
\begin{itemize}
    \item \textbf{Story:} ``My Teacher is a Robot''.
    \item \textbf{Visuals:} Robot pictures and simple videos.
    \item \textbf{Art:} Robot face templates and crayons.
    \item \textbf{Support:} Agree/disagree signs and open floor space.
\end{itemize}

\pamphletheading{Learning Outcomes}
\begin{tcolorbox}[colback=SmaileLightBlue, colframe=SmaileBlue, boxrule=0.5pt, arc=2mm]
\small
\textbf{Knowledge:}
\begin{itemize}[leftmargin=*]

\end{itemize}
\textbf{Skills:}
\begin{itemize}[leftmargin=*]
    \item Sharing ideas clearly.
    \item Listening and responding kindly.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=SmaileLightPurple, colframe=SmailePurple, boxrule=0.5pt, arc=2mm]
\small

    \item Kindness.

\pamphletheading{1. Story \& Sharing}
\textbf{Think--Pair--Share:}
Children listen to a robot teacher story, talk with a partner, and say what felt good, strange, kind, or unfair.

\pamphletheading{2. Mini-Debate}
\textbf{Agree or Disagree:}
Children respond to simple statements and give short reasons using sentence starters.

\pamphletheading{3. Creative Reflection}
\textbf{Robot Design:}
Children color a robot face and explain whether their robot is helpful, happy, fair, or unkind.
\begin{tcolorbox}[colback=white, colframe=SmaileGray, boxrule=0.5pt]
\small
\textbf{Discussion:}
``What can a robot do well, and what does a real teacher do better?''
\end{tcolorbox}
