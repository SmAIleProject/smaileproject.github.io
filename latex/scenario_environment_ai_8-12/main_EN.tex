% --- Activities ---
\section{The Invisible Cost of the Internet}
\textbf{Goal:} Make the invisible visible.
\begin{itemize}
    \item \textbf{Concept:} The "Cloud" isn't in the sky; it's in a building.
    \item \textbf{Activity:} Watch a video tour of a data center.
    \item \textbf{Discuss:} How much water and electricity does it take to cool those machines?
\end{itemize}

\section{My Digital Footprint}
\textbf{Goal:} Experience energy consumption.
\begin{itemize}
    \item \textbf{Role-Play:} Students act as servers. The teacher sends "requests" (cards) faster and faster.
    \item \textbf{Observation:} Servers heat up (get tired) and need "cooling" (fans/water).
    \item \textbf{Calculation:} Compare the footprint of sending an email vs. training an AI model.
\end{itemize}

\section{AI Eco-Builder}
\textbf{Goal:} Creative problem solving.
\begin{itemize}
    \item \textbf{Task:} Design a sustainable city.
    \item \textbf{Tools:} Minecraft, SimCity, or Art supplies.
    \item \textbf{Key Elements:}
    \begin{itemize}
        \item Efficient Transport (AI traffic lights).
        \item Green Energy (Solar/Wind).
        \item Waste Reduction (Recycling bots).
    \end{itemize}
    \item \textbf{Present:} Share the city design and explain the AI features.
\end{itemize}

\section{Reflection and Evaluation}
\textbf{Goal:} Personal responsibility.
\begin{itemize}
    \item \textbf{Self-Reflection:} Complete the reflection sheet (Att 3.1).
    \item \textbf{Discussion:} "What surprised you?" "What will you do differently?"
    \item \textbf{Action:} Pledge one eco-friendly digital habit (e.g., "I will delete old emails").
\end{itemize}
