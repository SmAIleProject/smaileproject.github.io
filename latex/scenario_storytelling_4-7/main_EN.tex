% --- Activities ---
\section{Lesson 1: The Big Idea}
\textbf{Goal:} Introduce storytelling and AI.
\begin{itemize}
    \item \textbf{Discussion:} "What makes a good story?" (Heroes, magic, places).
    \item \textbf{Structure:} Explain Start, Middle, End.
    \item \textbf{Meet Brainy:} Introduce the AI as a helper who has lots of ideas but needs children's imagination to make them real.
\end{itemize}

\section{Lesson 2: Our Adventure}
\textbf{Goal:} Collaborative creation.
\begin{itemize}
    \item \textbf{Step 1:} Groups pick 1 Character Card (e.g., Pirate) and 1 Setting Card (e.g., Space).
    \item \textbf{Step 2:} Teacher asks Brainy: "Brainy, give us a story starter for a Pirate in Space!"
    \item \textbf{Step 3:} Brainy gives a prompt (e.g., "The Pirate lost his treasure map on the moon!").
    \item \textbf{Step 4:} Children decide what happens next and draw it.
\end{itemize}

\section{Lesson 3: Sharing Time}
\textbf{Goal:} Presentation and feedback.
\begin{itemize}
    \item \textbf{Tell the Tale:} Groups show their drawings and tell the story.
    \item \textbf{Role Play:} Use funny voices or actions.
    \item \textbf{Feedback:} Use the "Giving Feedback to a Friend" form (smiley faces).
\end{itemize}

\section{Reflection and Evaluation}
\subsection*{My Story Checklist}
\begin{itemize}
    \item Did we have a beginning, middle, and end?
    \item Did we use Brainy's idea?
    \item Did we work together?
\end{itemize}

\subsection*{Teacher Evaluation}
\begin{itemize}
    \item \textbf{Structure:} Clear narrative parts.
    \item \textbf{Creativity:} Imaginative ideas.
    \item \textbf{Collaboration:} Teamwork quality.
    \item \textbf{AI Use:} Engagement with the tool.
\end{itemize}
