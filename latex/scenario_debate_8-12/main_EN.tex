% --- Activities ---
\section{Introduction and Motivation}
\textbf{Goal:} Understand the debate format.
\begin{itemize}
    \item \textbf{Structure:} Explain the roles using the Infographic.
    \item \textbf{Topics:}
    \begin{enumerate}
        \item "Should AI cameras monitor students in schools?"
        \item "Do Chatbot assistants make students less creative?"
    \end{enumerate}
\end{itemize}

\section{Research and Learning}
\textbf{Goal:} Build a strong case.
\begin{itemize}
    \item Teams conduct research to find facts and evidence.
    \item \textbf{Affirmative:} Focus on benefits (safety, efficiency).
    \item \textbf{Negative:} Focus on risks (privacy, dependency).
    \item Teams prepare their Opening Statements.
\end{itemize}

\section{The Debate (Creative Application)}
\textbf{Goal:} Present and persuade.
\begin{itemize}
    \item \textbf{Constructive Speeches (5 min):} Each team presents their main points.
    \item \textbf{Crossfire (5 min):} Teams ask each other questions.
    \item \textbf{Rebuttal \& Closing (5 min):} Respond to opponents and summarize.
\end{itemize}

\section{Reflection and Evaluation}
\textbf{Goal:} Assessment and feedback.
\begin{itemize}
    \item \textbf{Judges:} Use the Rubric to score arguments (30\%), evidence (25\%), and delivery (20\%).
    \item \textbf{Audience:} Share thoughts and vote on Padlet.
    \item \textbf{Self-Eval:} Students complete the reflection form.
\end{itemize}
