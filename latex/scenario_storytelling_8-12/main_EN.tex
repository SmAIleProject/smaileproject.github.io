% --- Activities ---
\section{Introduction and Motivation}
\textbf{Goal:} Engage students with the historical context.
\begin{itemize}
    \item \textbf{Watch:} Video "The Age of Exploration".
    \item \textbf{Discuss:} "If you could discover a new land today, where would you go and why?"
\end{itemize}

\section{Research and Learning}
\subsection{Explorer Profiles}
Students work in groups to create profiles for Columbus, Vasco da Gama, Magellan, or Vespucci.
\begin{itemize}
    \item Use AI tools to find: Background, Major Voyages, Discoveries.
    \item \textbf{Method:} Fishbowl Discussion to share findings.
\end{itemize}

\subsection{Creating Routes of Discoveries}
Students mark the route of their explorer on a map.
\begin{itemize}
    \item \textbf{Tool:} Google My Maps (collaborative online map).
    \item Analyze the significance of the places discovered.
\end{itemize}

\subsection{Why is Cinnamon in my Roll?}
\textbf{The Hook:} "If you lived in the Middle Ages, you would never have tasted chocolate."
\begin{itemize}
    \item Research foods brought to Europe from other continents.
    \item Use Padlet to sort foods by continent of origin.
    \item Discuss how these findings changed European cuisine.
\end{itemize}

\section{Creative Application}
\subsection{The Diary of an Explorer}
Students write diary entries or letters as if they were explorers.
\begin{itemize}
    \item Detail experiences, emotions, and fears.
    \item \textbf{AI Task:} Use AI image generators to create illustrations that match their specific diary descriptions.
\end{itemize}

\subsection{Interactive Visual Storytelling}
Students prepare a presentation of their diary entries.
\begin{itemize}
    \item \textbf{Hook:} "Have you ever sailed on a Carrack?"
    \item Use the AI-generated visuals to bring the story to life.
\end{itemize}
