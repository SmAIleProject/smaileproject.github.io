% --- Activities ---
\section{Introduction: AI Failures}
\textbf{Goal:} Understand AI isn't perfect.
\begin{itemize}
    \item \textbf{Video:} "Funny Object Recognition Fails".
    \item \textbf{Discussion:} "Which mistake surprised you?"
    \item \textbf{Concept:} AI is only as good as the data you give it.
\end{itemize}

\section{Activity 1: Many Faces, One World}
\textbf{Goal:} Compare human and machine vision.
\begin{itemize}
    \item \textbf{Task:} Group photos of diverse people based on criteria.
    \item \textbf{Discussion:} Humans see "culture" or "gender"; machines see "pixels" and "patterns".
\end{itemize}

\section{Activity 2: Exploring Traffic Signs}
\textbf{Goal:} Data collection.
\begin{itemize}
    \item \textbf{Field Work:} Photograph local traffic signs.
    \item \textbf{Collaboration:} Upload to eTwinning to share with partner schools.
    \item \textbf{Analysis:} Are "Stop" signs the same in every country?
\end{itemize}

\section{Activity 3: Training the AI Model}
\textbf{Goal:} Practical Machine Learning.
\begin{itemize}
    \item \textbf{Tool:} Teachable Machine.
    \item \textbf{Step 1:} Create classes (e.g., "Warning Signs", "Regulatory Signs").
    \item \textbf{Step 2:} Upload images and train.
    \item \textbf{Step 3:} Test with a sign from a partner country. Did it work?
\end{itemize}

\section{Activity 4: Ethics \& Reflection}
\textbf{Goal:} Critical thinking.
\begin{itemize}
    \item \textbf{Debate:} "Can AI replace a lab scientist?"
    \item \textbf{Reflection:} Share thoughts on Padlet.
    \item \textbf{Output:} Digital reflection on the importance of human judgment.
\end{itemize}
