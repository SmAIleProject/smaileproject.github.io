% --- Outcomes ---
\section*{Learning Outcomes}
By the end of the project, students will be able to:

\subsubsection*{KNOWLEDGE \& UNDERSTANDING:}
\begin{itemize}
    \item Recognize artistic styles and understand how AI mimics them.
    \item Understand the relationship between visual art and musical atmosphere (tempo, mood).
\end{itemize}

\subsubsection*{SKILLS \& ABILITIES:}
\begin{itemize}
    \item Use advanced digital tools (Canva, AI Generators) for multimedia creation.
    \item Curate a cohesive digital exhibition presentation.
    \item Compare human-made art with AI-generated versions critically.
\end{itemize}

\subsubsection*{ATTITUDES \& VALUES:}
\begin{itemize}
    \item Appreciate the ethical complexities of AI in art.
    \item Value collaboration and constructive peer feedback.
\end{itemize}

% --- Activities ---
\section{Research \& Creation (Weeks 1-2)}
\textbf{Goal:} The Human vs. The Machine.
\begin{itemize}
    \item \textbf{Human Art:} Students create a digital piece inspired by a specific movement (e.g., Surrealism).
    \item \textbf{AI Art:} Students write prompts to generate a similar piece.
    \item \textbf{Comparison:} Analyze differences in emotion, technique, and "soul".
\end{itemize}

\section{Music Integration (Week 3)}
\textbf{Goal:} Multi-sensory experience.
\begin{itemize}
    \item \textbf{Task:} Generate a background track for the artwork.
    \item \textbf{Experimentation:} Adjust variables like Tempo (fast/slow), Instruments (electronic/acoustic), and Mood.
    \item \textbf{Selection:} Justify why the chosen track fits the visual art.
\end{itemize}

\section{The Exhibition (Week 4)}
\textbf{Goal:} Professional presentation.
\begin{itemize}
    \item \textbf{Digital Presentation:} Create a slide deck/video in Canva including:
    \begin{itemize}
        \item Artist/Style context.
        \item Human vs. AI artwork comparison.
        \item Integrated background music.
        \item Reflection on the creative process.
    \end{itemize}
    \item \textbf{Gallery Walk:} Share on Padlet. Students leave comments and "likes".
    \item \textbf{Assessment:} Complete Group Reflection and Self-Assessment Rubrics.
\end{itemize}
