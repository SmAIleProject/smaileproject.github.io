% --- Outcomes ---
\section*{Learning Outcomes}
By the end of the project, students will be able to:

\subsubsection*{KNOWLEDGE \& UNDERSTANDING:}
\begin{itemize}
    \item Explain "CO$_2$ equivalents" and data center energy consumption.
    \item Understand the energy and water costs involved in AI operations.
\end{itemize}

\subsubsection*{SKILLS \& ABILITIES:}
\begin{itemize}
    \item Compare energy costs of digital actions (email vs. AI training).
    \item Work in teams to brainstorm sustainable solutions.
    \item Present sustainable AI-based ideas clearly.
\end{itemize}

\subsubsection*{ATTITUDES \& VALUES:}
\begin{itemize}
    \item Demonstrate responsibility for digital habits.
    \item Appreciate the role of innovation in sustainability.
\end{itemize}

% --- Activities ---
\section{Introduction: The Invisible Cost}
\textbf{Goal:} Visualize the physical internet.
\begin{itemize}
    \item \textbf{Activity:} Watch a video about data centers.
    \item \textbf{Discuss:} "Where does the 'Cloud' live?" "Why do servers need cooling?"
\end{itemize}

\section{Research: Guess the Impact}
\textbf{Goal:} Understand carbon footprints.
\begin{itemize}
    \item \textbf{Task:} Students research the emissions of various activities.
    \item \textbf{Comparison:} Watching HD video vs. Sending an email vs. Training a large AI model.
    \item \textbf{Tool:} Use ChatGPT to find estimates.
\end{itemize}

\section{Creative: AI Solutions Mission}
\textbf{Goal:} Innovation for sustainability.
\begin{itemize}
    \item \textbf{Group Work:} Teams invent 2 AI-based solutions to reduce impact.
    \item \textbf{Prompts:}
    \begin{itemize}
        \item Smart AI to optimize home energy.
        \item AI to reduce food waste.
        \item AI ride-sharing algorithms.
    \end{itemize}
    \item \textbf{Output:} Create a collaborative Mind Map.
\end{itemize}

\section{Reflection and Evaluation}
\textbf{Goal:} Personal responsibility.
\begin{itemize}
    \item \textbf{Presentations:} Teams share their solutions.
    \item \textbf{Padlet Reflection:} "What surprised me?" "What action will I take?"
    \item \textbf{Assessment:} Complete the final student workshop evaluation.
\end{itemize}
