\begin{itemize}

    \item Learn that AI can help with tasks but needs human guidance.

% --- Activities ---
\section{Introduction and Motivation}
\subsection{Story: The Robot Who Got It Wrong}
\textbf{Goal:} Spark curiosity and show that robots need help.
\begin{itemize}
    \item \textbf{Activity:} Teacher reads a story about a robot who gets confused (e.g., putting shoes in the fridge).
    \item \textbf{Discussion:} "Why did the robot get it wrong?" "How do people know better?"
    \item \textbf{Key Concept:} Robots are helpful but don't think like people.
\end{itemize}

\section{Research and Learning}
\subsection{Exploring Real AI Helpers}
\textbf{Goal:} Connect concepts to real life.
\begin{itemize}
    \item \textbf{Video:} Watch a short clip about robots (vacuums, delivery bots).
    \item \textbf{Discussion:} "Can a robot vacuum find your lost socks?" (No). "Can a smart speaker give you a hug?" (No).
    \item \textbf{Voting Game:} Teacher shows cards (e.g., Hospital Robot). Children vote if it needs a person to help.
\end{itemize}

\section{Creative Application}
\subsection{Robot Helper or Human Job?}
\textbf{Sorting Game:}
\begin{itemize}
    \item Teacher calls out tasks: "Tidy room", "Give a hug", "Paint a picture".
    \item Children move to zones: \textbf{Robot}, \textbf{Person}, or \textbf{Both}.
\end{itemize}

\subsection{Drawing Task: Design Your Robot}
\textbf{Assignment:} Invent your own robot helper.
\begin{itemize}
    \item What does it do well? (e.g., carry heavy boxes).
    \item What does it need help with? (e.g., knowing where to go).
\end{itemize}

\section{Reflection and Evaluation}
\subsection{Circle Time \& Memory Box}
\begin{itemize}
    \item Children share their robot drawings.
    \item \textbf{Memory Box:} Add drawings/notes of "How AI helped today" (e.g., Mom used a map app).
\end{itemize}

\subsection*{Assessment Checklist}
\begin{itemize}
    \item Recognised robot roles vs. human roles.
    \item Described how their robot helps.
    \item Engaged with drawing/memory box.
    \item Shared ideas confidently.
\end{itemize}
