% --- Outcomes ---
\section*{Learning Outcomes}
By the end of the project, children will be able to:

\subsubsection*{KNOWLEDGE \& UNDERSTANDING:}
\begin{itemize}
    \item Recognize styles/colors of famous artists (e.g., Van Gogh, Kandinsky).
    \item Understand that AI computers can help make pictures.
    \item Understand how music connects to visual feelings.
\end{itemize}

\subsubsection*{SKILLS \& ABILITIES:}
\begin{itemize}
    \item Use drawing apps (Paint/Canva) to create simple art.
    \item Explore AI tools to generate images and compare results.
    \item Talk about their art and explain color choices.
\end{itemize}

\subsubsection*{ATTITUDES \& VALUES:}
\begin{itemize}
    \item Develop cultural curiosity about art.
    \item Cooperate and show respect for others' work.
    \item Feel pride in showcasing their creations.
\end{itemize}

% --- Activities ---
\section{Week 1: Introduction to Art and AI}
\textbf{Goal:} Spark curiosity about color, feelings, and machines.
\begin{itemize}
    \item \textbf{Art Slideshow:} Show works by Van Gogh or Miró. Ask: "What colors do you see?" "Is it happy or sad?"
    \item \textbf{Video:} Watch a short "What is AI?" clip. Discuss: "Can a robot paint?"
    \item \textbf{Activity:} Art Naming Game – children invent fun titles for famous paintings.
\end{itemize}

\section{Week 2: Drawing Like Artists}
\textbf{Goal:} Express ideas using digital tools.
\begin{itemize}
    \item \textbf{Digital Demo:} Teacher shows how to use simple brushes in Canva/Paint.
    \item \textbf{Create:} Children make their own version of a famous painting style.
    \item \textbf{AI Collaboration:} Children describe their drawing to the teacher, who generates an AI version.
    \item \textbf{Compare:} "What is the same? What is different?"
\end{itemize}

\section{Week 3: Art and Music Together}
\textbf{Goal:} Connect sound to visual art.
\begin{itemize}
    \item \textbf{Listening Game:} Play happy/sad/spooky music. Ask: "What color is this sound?"
    \item \textbf{Match:} Children choose a music clip that matches their artwork.
    \item \textbf{Optional AI Music:} Teacher generates a simple tune based on the child's description (e.g., "Yellow sun music").
\end{itemize}

\section{Week 4: Sharing and Reflecting}
\textbf{Goal:} Celebrate achievements.
\begin{itemize}
    \item \textbf{Preparation:} Teacher collects drawings and music into a slideshow. Children record a short voice message: "This is my picture..."
    \item \textbf{Gallery Walk:} The class watches the exhibition together.
    \item \textbf{Reflection Circle:} "What did you like best?" "Who was more creative, you or the AI?"
\end{itemize}
