% --- Activities ---
\section{Introduction: AI in Science}
\textbf{Goal:} Contextualize AI as a tool.
\begin{itemize}
    \item \textbf{Discussion:} How does AI identify a star or a cell?
    \item \textbf{Concept:} It learns from examples provided by humans.
\end{itemize}

\section{Activity 1: The Human Algorithm}
\textbf{Goal:} Understand classification logic.
\begin{itemize}
    \item \textbf{Task:} Students are given a set of objects (e.g., various rocks).
    \item \textbf{Action:} Create rules to classify them manually.
    \item \textbf{Reflection:} How did you decide? What features did you look for?
\end{itemize}

\section{Activity 2: Collecting Data}
\textbf{Goal:} Building a robust dataset.
\begin{itemize}
    \item \textbf{Task:} Teams capture photos of their assigned category.
    \item \textbf{Focus:} Quality and diversity (lighting, background) to avoid bias.
    \item \textbf{Output:} A folder of labeled images.
\end{itemize}

\section{Activity 3: Training the Model}
\textbf{Goal:} Practical Machine Learning.
\begin{itemize}
    \item \textbf{Tool:} Teachable Machine.
    \item \textbf{Process:} Upload data -> Train Model -> Test.
    \item \textbf{Analysis:} Present new objects. Does the AI recognize them? Identify false positives/negatives.
\end{itemize}

\section{Activity 4: Ethics \& Debate}
\textbf{Goal:} Critical thinking.
\begin{itemize}
    \item \textbf{Debate Topic:} "Can AI replace human lab scientists?"
    \item \textbf{Discussion:} Reliability, responsibility for errors, and the need for human oversight.
    \item \textbf{Evaluation:} Students reflect on their learning via Padlet and peer feedback forms.
\end{itemize}
