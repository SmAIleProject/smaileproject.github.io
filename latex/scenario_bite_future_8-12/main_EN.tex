% --- Outcomes ---
\section*{Learning Outcomes}
Students will be able to:

\subsubsection*{KNOWLEDGE \& UNDERSTANDING:}
\begin{itemize}
    \item Explain why food is important for health and sustainability.
    \item Describe different ways food is grown (e.g., aquaponics, vertical farming).
    \item Recognize that some foods (insects, algae) might be common in the future.
\end{itemize}

\subsubsection*{SKILLS \& ABILITIES:}
\begin{itemize}
    \item Look for simple information about food technology.
    \item Share ideas and work collaboratively with others.
    \item Communicate ideas by drawing and talking.
\end{itemize}

\subsubsection*{ATTITUDES \& VALUES:}
\begin{itemize}
    \item Show curiosity about new things.
    \item Think about being responsible with food and taking care of the planet.
    \item Listen to other people's ideas.
\end{itemize}

% --- Activities ---
\section{Strange Food of the Future}
\textbf{Goal:} Investigate sustainable innovations.
\begin{itemize}
    \item \textbf{Hook:} "Imagine it is 2050. What will we eat if there is less land for farming?"
    \item \textbf{Group Work:} Divide students into teams. Each team researches one topic:
    \begin{itemize}
        \item Edible Insects.
        \item Vertical Farming.
        \item Lab-Grown Meat.
        \item 3D Printed Food.
    \end{itemize}
    \item \textbf{Task:} Find 3 facts and 1 benefit for the planet.
\end{itemize}

\section{My Futuristic Menu}
\textbf{Goal:} Apply knowledge creatively.
\begin{itemize}
    \item \textbf{Challenge:} Design a lunch menu for a restaurant in 2050.
    \item \textbf{Requirements:}
    \begin{itemize}
        \item Must be healthy.
        \item Must use sustainable ingredients (e.g., Algae, Insects, Plant-based).
        \item Must look delicious (Drawings/Collage).
    \end{itemize}
    \item \textbf{Sharing:} Present the menu to the class.
\end{itemize}

\section{Reflection and Evaluation}
\textbf{Goal:} Assess understanding and teamwork.
\begin{itemize}
    \item \textbf{Self-Evaluation:} Students complete the reflection sheet (Att 4.1.1).
    \item \textbf{Teacher Rubric:} Evaluate creativity and collaboration (Att 4.1.2).
    \item \textbf{Discussion:} "Which future food would you try?"
\end{itemize}
