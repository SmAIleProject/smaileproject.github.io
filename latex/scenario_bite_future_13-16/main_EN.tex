% --- Activities ---
\section{Introduction and Motivation}
\textbf{Goal:} Introduce the theme of future food and activate prior knowledge.
\begin{itemize}
    \item \textbf{Opening Context:} Climate change, pollution, and food waste require new ideas about how we produce and consume food.
    \item \textbf{Video Prompt:} Students watch the short video ``Future Food''.
    \item \textbf{Guided Questions:}
    \begin{itemize}
        \item Would you eat something unusual, such as insects or lab-grown meat, if it could help the planet?
        \item What seemed the strangest or most interesting idea in the video?
        \item Do you think we will need to change how we eat in the future?
    \end{itemize}
    \item \textbf{Transition:} Students prepare to research and imagine their own ``Menu of Tomorrow''.
\end{itemize}

\section{Activity 1: Research and Learning}
\textbf{Goal:} Investigate future food technologies critically.
\begin{itemize}
    \item \textbf{Team Formation:} Students draw an image from a box to join a topic team.
    \item \textbf{Research Topics:}
    \begin{itemize}
        \item Vertical Farming
        \item Lab-Grown Meat
        \item Insect-Based Nutrition
        \item Aquaponics and Hydroponics
    \end{itemize}
    \item \textbf{Worksheet Questions:}
    \begin{itemize}
        \item What is it?
        \item What problem does it solve?
        \item What are its advantages and disadvantages?
        \item Does it already exist locally or globally?
        \item Would you use it? Why or why not?
        \item What does ChatGPT or another chatbot say about it?
        \item Are the AI answers consistent with expert sources?
    \end{itemize}
    \item \textbf{Research Expectation:} Students use reliable sources and explain their reasoning clearly.
\end{itemize}

\section{Activity 2: Creative Application}
\textbf{Goal:} Turn research into a structured digital product.
\begin{itemize}
    \item \textbf{Task:} Teams create an e-book in Book Creator or Canva.
    \item \textbf{Suggested e-book structure:}
    \begin{itemize}
        \item Title
        \item What it is
        \item How it works
        \item Where it already exists
        \item Positive and negative aspects
        \item Would you use it?
        \item AI opinion compared with expert sources
        \item Message to society
        \item Literature
    \end{itemize}
    \item \textbf{Design Requirement:} Include at least one image, ideally created with an AI image tool.
    \item \textbf{Presentation:} Each team presents its e-book to the class and receives peer assessment.
\end{itemize}

\section{Activity 3: Reflection and Evaluation}
\textbf{Goal:} Consolidate learning and evaluate the workshop.
\begin{itemize}
    \item \textbf{Brainstorming:} Students design an ideal ``Future Lunch'' with a warm starter, main course, and dessert.
    \item \textbf{Discussion:} Students compare menus and comment on each other's ideas using Padlet or classroom discussion.
    \item \textbf{Assessment:} Final student success combines peer evaluation and teacher assessment.
    \item \textbf{Final Evaluation:} Students complete a workshop evaluation form using paper or digital tools such as Google Forms or Microsoft Forms.
\end{itemize}
