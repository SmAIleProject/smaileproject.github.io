    \end{description}
    \textbf{Key Learning Goals:}
    \begin{enumerate}
        \item \textbf{Machine Learning Concept:} Understand that computers learn from examples, just like people.
        \item \textbf{Role of Humans:} Learn that computers need humans to show them examples (data).
        \item \textbf{Mistakes:} Notice that sometimes machines get things wrong if they are confused.
        \item \textbf{Categorization:} Practice sorting and grouping images to "teach" the robot.
    \end{enumerate}
\end{keyinfobox}

% --- Outcomes ---
\section*{Learning Outcomes}
By the end of the project, students will be able to:

\subsubsection*{KNOWLEDGE \& UNDERSTANDING:}
\begin{itemize}
    \item Know that machines can "look" and "learn" from pictures.
    \item Understand that computers need humans to show them examples.
    \item Notice that sometimes machines get things wrong.
\end{itemize}

\subsubsection*{SKILLS \& ABILITIES:}
\begin{itemize}
    \item Sort and group images (e.g., Animals vs. Birds, Fruits vs. Vegetables).
    \item Describe what they see in pictures.
    \item Notice patterns or differences between objects.
\end{itemize}

\subsubsection*{ATTITUDES \& VALUES:}
\begin{itemize}
    \item Feel curious about how robots work.
    \item Learn to share ideas and listen to others.
    \item Recognize that people and robots can work together.
\end{itemize}

% --- Activities ---
\section{Introduction: Train the Robot}
\textbf{Concept:} Machine Learning is like teaching a child with examples.
\begin{itemize}
    \item \textbf{Hook:} "I have a robot friend who doesn't know what a strawberry is! Can we teach him?"
    \item \textbf{Activity:} Place hoops labelled "Fruit" and "Vegetable".
    \item \textbf{Action:} Students sort cards into the hoops.
    \item \textbf{Explanation:} "You are giving the robot \textit{data} so it can learn patterns."
\end{itemize}

\section{Testing: Robot Guesses}
\textbf{Concept:} Testing the model and handling errors.
\begin{itemize}
    \item \textbf{Role-Play:} Teacher puts on a robot voice.
    \item \textbf{Scenario:} Robot picks up a tomato card and says "Beep boop... this is an Apple!"
    \item \textbf{Response:} Children shout "No!" and explain why (e.g., "It's a vegetable/fruit," "It's red but different shape").
    \item \textbf{Lesson:} Robots make mistakes if they haven't seen enough examples.
\end{itemize}

\section{Advanced: Human vs. Robot}
\textbf{Concept:} Emotional Intelligence vs. Artificial Intelligence.
\begin{itemize}
    \item \textbf{Cards:} Use "Action Cards" (e.g., Adding numbers, Giving a hug, Knowing red).
    \item \textbf{Sorting:} "Can a robot do this, or do we need a person?"
    \item \textbf{Discussion:} Use the Emotion Cards. Can a robot feel "Happy" or just show a picture of a smile?
\end{itemize}

\section{Reflection: Creative Drawing}
\textbf{Task:} "I Helped the Robot!"
\begin{itemize}
    \item Students draw a picture of themselves teaching the robot.
    \item \textbf{Captions:} Teacher helps scribe: "I showed the robot a green apple" or "I told it the dog was not a cat".
    \item \textbf{Display:} Create a "Robot Helpers" wall in the classroom.
\end{itemize}
